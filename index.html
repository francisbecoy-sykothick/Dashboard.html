const SECRET = "YOUR_TOKEN_HERE";
const ADMIN_EMAIL = "your-email@example.com"; // Change this to your email

function doGet(e) {
  if (e.parameter.token !== SECRET)
    return resp(JSON.stringify({success:false,message:"Unauthorized"}),true);

  const action = e.parameter.action || "read";
  const ss = SpreadsheetApp.getActiveSpreadsheet();

  // LOGIN
  if (action === "login") {
    const loginSheet = ss.getSheetByName("Login");
    if (!loginSheet)
      return resp(JSON.stringify({success:false,message:"Login sheet not found"}),true);
    const rows = loginSheet.getDataRange().getValues();
    const user = e.parameter.username || "";
    const pass = e.parameter.password || "";
    for (let i = 1; i < rows.length; i++) {
      if (String(rows[i][0]).trim() === user &&
          String(rows[i][1]).trim() === pass)
        return resp(JSON.stringify({success:true}),true);
    }
    return resp(JSON.stringify({success:false,message:"Incorrect username or password."}),true);
  }

  // GET PENDING
  if (action === "getPending") {
    const ps = ss.getSheetByName("Pending");
    if (!ps) return resp(JSON.stringify([]),true);
    const rows = ps.getDataRange().getValues();
    return resp(JSON.stringify(rows),true);
  }

  // GET MY PENDING
  if (action === "getMyPending") {
    const ps = ss.getSheetByName("Pending");
    if (!ps) return resp(JSON.stringify([]),true);
    const rows = ps.getDataRange().getValues();
    const user = e.parameter.username || "";
    const filtered = rows.filter((r,i) => i===0 || String(r[4]).trim()===user);
    return resp(JSON.stringify(filtered),true);
  }

  // GET ACTIVITY LOG
  if (action === "getActivity") {
    const al = ss.getSheetByName("ActivityLog");
    if (!al) return resp(JSON.stringify([]),true);
    const rows = al.getDataRange().getValues();
    return resp(JSON.stringify(rows),true);
  }

  // READ main data
  const sheet = ss.getActiveSheet();
  const rows = sheet.getDataRange().getValues();
  return ContentService
    .createTextOutput(JSON.stringify(rows))
    .setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  const data = JSON.parse(e.postData.contents);
  if (data.token !== SECRET)
    return resp("Unauthorized");

  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const action = data.action || "write";

  // SUBMIT PENDING
  if (action === "submitPending") {
    let ps = ss.getSheetByName("Pending");
    if (!ps) {
      ps = ss.insertSheet("Pending");
      ps.appendRow(["ID","Date","Name","Amount","SubmittedBy","Status","SubmittedAt"]);
    }
    const id = "P" + Date.now();
    ps.appendRow([
      id, data.date, data.name, data.amount,
      data.submittedBy || "", "Pending", new Date().toISOString()
    ]);
    
    // Log activity
    logActivity(ss, data.submittedBy || "User", "Submit", 
      `Submitted ₱${data.amount} for ${data.name}`, id);
    
    // Send email notification
    try {
      MailApp.sendEmail({
        to: ADMIN_EMAIL,
        subject: "⏳ New Pending Approval - Copra Trade",
        body: `A new transaction has been submitted for approval:

Name: ${data.name}
Amount: ₱${data.amount}
Submitted by: ${data.submittedBy}
Date: ${data.date}

Log in to Dashboard to approve or reject.`
      });
    } catch(err) { Logger.log("Email failed: " + err); }
    
    return resp(JSON.stringify({success:true,id:id}),true);
  }

  // APPROVE PENDING
  if (action === "approvePending") {
    const ps = ss.getSheetByName("Pending");
    if (!ps) return resp("No Pending sheet");
    const rows = ps.getDataRange().getValues();
    for (let i = 1; i < rows.length; i++) {
      if (String(rows[i][0]) === String(data.id)) {
        ps.getRange(i+1, 6).setValue("Approved");
        const mainSheet = ss.getActiveSheet();
        mainSheet.appendRow([
          rows[i][1], rows[i][2], "Release", rows[i][3],
          "", new Date().toISOString(), ""
        ]);
        logActivity(ss, data.user || "Admin", "Approve", 
          `Approved ₱${rows[i][3]} for ${rows[i][2]}`, data.id);
        return resp(JSON.stringify({success:true}),true);
      }
    }
    return resp(JSON.stringify({success:false,message:"ID not found"}),true);
  }

  // REJECT PENDING
  if (action === "rejectPending") {
    const ps = ss.getSheetByName("Pending");
    if (!ps) return resp("No Pending sheet");
    const rows = ps.getDataRange().getValues();
    for (let i = 1; i < rows.length; i++) {
      if (String(rows[i][0]) === String(data.id)) {
        ps.getRange(i+1, 6).setValue("Rejected");
        logActivity(ss, data.user || "Admin", "Reject", 
          `Rejected ₱${rows[i][3]} for ${rows[i][2]}`, data.id);
        return resp(JSON.stringify({success:true}),true);
      }
    }
    return resp(JSON.stringify({success:false,message:"ID not found"}),true);
  }

  // LOG ACTIVITY (manual log from Dashboard)
  if (action === "logActivity") {
    logActivity(ss, data.user || "Unknown", data.actionType || "Action", 
      data.details || "", data.txId || "");
    return resp(JSON.stringify({success:true}),true);
  }

  // Default WRITE
  const sheet = ss.getActiveSheet();
  sheet.appendRow([
    data.date, data.name, data.type,
    data.amount, "", new Date().toISOString(),
    data.notes || ""
  ]);
  
  // Log activity
  logActivity(ss, data.user || "Admin", "Add", 
    `Added ${data.type} ₱${data.amount} for ${data.name}`, "");
  
  return resp("OK");
}

function logActivity(ss, user, action, details, txId) {
  let al = ss.getSheetByName("ActivityLog");
  if (!al) {
    al = ss.insertSheet("ActivityLog");
    al.appendRow(["Timestamp","User","Action","Details","TransactionID"]);
  }
  al.appendRow([new Date().toISOString(), user, action, details, txId]);
}

function resp(msg, json) {
  const out = ContentService.createTextOutput(
    typeof msg === "string" ? msg : JSON.stringify(msg)
  );
  if (json) out.setMimeType(ContentService.MimeType.JSON);
  return out;
}
